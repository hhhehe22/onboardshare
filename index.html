function digest_(value) {

/** 누구나 호출 가능한 공개 함수이므로 일회용 설정 코드를 반드시 확인합니다. */
function initializeAdmin(setupCode, password, confirmation) {
  if (typeof password !== 'string' || password.length < 12 || password.length > 128) {
    throw new Error('관리자 비밀번호는 12~128자로 설정해 주세요.');
  }
  if (password !== confirmation) throw new Error('비밀번호 확인이 일치하지 않습니다.');
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const props = PropertiesService.getScriptProperties();
    if (props.getProperty(ADMIN_HASH_KEY)) throw new Error('이미 관리자 비밀번호가 설정되어 있습니다.');
    const tokenHash = props.getProperty(SETUP_HASH_KEY);
    const expires = Number(props.getProperty(SETUP_EXPIRES_KEY)||0);
    if (!tokenHash || Date.now()>expires) throw new Error('초기 설정 코드가 없습니다. 스프레드시트의 ON BOARD 관리 메뉴에서 새 코드를 발급하세요.');
    if (typeof setupCode !== 'string' || digest_(setupCode.trim()) !== tokenHash) throw new Error('초기 설정 코드가 올바르지 않습니다.');
    const salt = Utilities.getUuid()+'-'+Utilities.getUuid();
    props.setProperties({[ADMIN_HASH_KEY]:digest_(salt+':'+password),[ADMIN_SALT_KEY]:salt});
    props.deleteProperty(SETUP_HASH_KEY);
    props.deleteProperty(SETUP_EXPIRES_KEY);
    return {configured:true};
  } finally {lock.releaseLock();}
}

function saveBoard(password, payload) {

  if (payload.pages.length < 1 || payload.pages.length > MAX_PAGES) throw new Error('페이지는 1~30개까지 가능합니다.');

    const title = String(p.title || '').trim().slice(0, 120);
    const content = String(p.content || '');
    const fontSize = Math.min(72, Math.max(14, Number(p.fontSize) || 26));

  });
  const ids = pages.map(function(p) { return p.id; });

/** 관리자가 이미지를 업로드하면 현재 스프레드시트의 상위 폴더에 있는 이미지 전용 하위 폴더에 저장합니다. */
function uploadImage(password, name, mimeType, base64Data) {
  checkPassword_(password);
  if (!/^image\/(jpeg|png|webp)$/.test(String(mimeType))) throw new Error('JPG, PNG, WEBP 이미지만 업로드할 수 있습니다.');
  if (typeof base64Data !== 'string' || base64Data.length > 1350000 || !/^[A-Za-z0-9+/=]+$/.test(base64Data)) {
    throw new Error('이미지 데이터가 너무 크거나 올바르지 않습니다.');
  }
  const bytes = Utilities.base64Decode(base64Data);
  if (bytes.length > MAX_IMAGE_BYTES) throw new Error('이미지가 너무 큽니다. 더 작게 압축해서 업로드하세요.');
  const safeName = String(name || 'onboard-image').replace(/[^\w.\-가-힣]/g,'_').slice(0,70);
  const blob = Utilities.newBlob(bytes,mimeType,safeName);
  const file = mediaFolder_().createFile(blob);
  // Anyone-with-link 공유 권한을 부여하지 않습니다. Drive 원본은 비공개입니다.
  return {id:file.getId(),name:safeName};
}

/** 읽기 전용 방문자는 게시된 이미지 ID만 조회 가능. */
function getBoardImage(id) {
  id = String(id || '');
  if (!/^[\w-]{15,110}$/.test(id)) throw new Error('잘못된 이미지 식별자입니다.');
  const board = readBoard_();
  const marker = '[[image:' + id + ']]';
  if (!board.pages.some(function(p) { return p.content.indexOf(marker) !== -1 || (p.elements || []).some(function(item) { return item.type === 'image' && item.imageId === id; }); })) {
    throw new Error('공유되지 않은 이미지입니다.');
  }
  const blob = DriveApp.getFileById(id).getBlob();
  const mime = blob.getContentType();
  if (!/^image\/(jpeg|png|webp)$/.test(mime)) throw new Error('허용되지 않은 이미지 형식입니다.');
  if (blob.getBytes().length > MAX_IMAGE_BYTES) throw new Error('이미지가 너무 큽니다.');
  return 'data:' + mime + ';base64,' + Utilities.base64Encode(blob.getBytes());
}

function checkPassword_(password) {
  const props = PropertiesService.getScriptProperties();
  const hash = props.getProperty(ADMIN_HASH_KEY);

    throw new Error('관리자 비밀번호가 올바르지 않습니다.');
  }
}

/**
 * 이미지 전용 폴더를 현재 스프레드시트 파일의 상위 폴더 안에 만듭니다.
 * 이미 생성된 폴더는 재사용하며, 스프레드시트를 다른 폴더로 옮기면
 * 이후의 업로드부터 새 위치에 있는 전용 폴더를 사용합니다.
 */
function mediaFolder_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error('스프레드시트에 연결된 Apps Script에서 실행해야 합니다.');
  const spreadsheetFile = DriveApp.getFileById(ss.getId());
  const parents = spreadsheetFile.getParents();
  // 최상위 My Drive에 저장된 스프레드시트도 부모 폴더가 있습니다.
  if (!parents.hasNext()) throw new Error('스프레드시트의 상위 폴더를 찾을 수 없습니다.');
  const parent = parents.next();
  const propertyKey = 'ONBOARD_MEDIA_FOLDER_IN_PARENT_' + ss.getId();
  const props = PropertiesService.getScriptProperties();
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const storedId = props.getProperty(propertyKey);
    if (storedId) {
      try {
        const previous = DriveApp.getFolderById(storedId);
        const previousParents = previous.getParents();
        while (previousParents.hasNext()) {
          if (previousParents.next().getId() === parent.getId()) return previous;
        }
      } catch (e) { /* 삭제하거나 이동된 폴더는 새로 찾습니다. */ }
    }
    const folderName = 'ON BOARD 업로드 이미지';
    const folders = parent.getFoldersByName(folderName);
    const folder = folders.hasNext() ? folders.next() : parent.createFolder(folderName);
    props.setProperty(propertyKey, folder.getId());
    return folder;
  } finally {
    lock.releaseLock();
  }
}

최초 1회 권한 검토 및 승인 (경고창 해결)

최초 배포 시 구글 보안 경고창이 나타날 수 있습니다. 이는 비공인 개인 스크립트이므로 정상적인 과정입니다:

  1. [액세스 승인] 버튼 클릭

  3. "Google에서 확인하지 않은 앱" 화면에서 좌측 하단 [고급 (Advanced)] 클릭
